// 화면 크기 및 래퍼(Wrapper)의 스크롤 진행도를 추적하는 커스텀 훅
const useScrollAndWindow = () => {
  const [windowInfo, setWindowInfo] = useState({ width: 1000, height: 800 });
  const [scrollY, setScrollY] = useState(0);
  const scrollRef = useRef(null);

  useEffect(() => {
    const handleResize = () => setWindowInfo({ width: window.innerWidth, height: window.innerHeight });
    const handleScroll = () => {
      if (scrollRef.current) {
        setScrollY(scrollRef.current.scrollTop);
      }
    };

    handleResize();
    handleScroll();

    window.addEventListener('resize', handleResize);
    const scrollEl = scrollRef.current;
    if (scrollEl) {
      scrollEl.addEventListener('scroll', handleScroll, { passive: true });
    }

    return () => {
      window.removeEventListener('resize', handleResize);
      if (scrollEl) {
        scrollEl.removeEventListener('scroll', handleScroll);
      }
    };
  }, []);

  return { ...windowInfo, scrollY, scrollRef };
};
